cmake_minimum_required(VERSION 3.25)
project(DreamGPUGuest LANGUAGES C CXX)
get_filename_component(DREAMGPU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DREAMGPU_GUEST "${CMAKE_CURRENT_SOURCE_DIR}")
set(DREAMGPU_CMAKE "${DREAMGPU_ROOT}/support/guest/cmake")
set(DREAMGPU_GUEST_OS "" CACHE STRING "win98 or nt5")
set_property(CACHE DREAMGPU_GUEST_OS PROPERTY STRINGS win98 nt5)
if(NOT DREAMGPU_GUEST_OS MATCHES "^(win98|nt5)$")
  message(FATAL_ERROR "DREAMGPU_GUEST_OS must be win98 or nt5")
endif()
if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Guest artifacts require the pinned MinGW GCC/G++ toolchain")
endif()
if(NOT CMAKE_C_COMPILER_VERSION VERSION_EQUAL "16.1.1")
  message(FATAL_ERROR "Guest artifacts require GCC16.1.1; got ${CMAKE_C_COMPILER_VERSION}")
endif()
option(DREAMGPU_BUILD_TRANSLATORS "Build pinned OpenGLide and WineD3D" ON)
option(DREAMGPU_BUILD_TOOLS "Build reusable controls/probes separately from runtime" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include("${DREAMGPU_CMAKE}/runtime.cmake")
include("${DREAMGPU_CMAKE}/opengl.cmake")
set(DREAMGPU_RUNTIME_TARGETS dgpugl)
include("${DREAMGPU_CMAKE}/${DREAMGPU_GUEST_OS}.cmake")
if(DREAMGPU_BUILD_TRANSLATORS)
  include("${DREAMGPU_CMAKE}/glide.cmake")
  include("${DREAMGPU_CMAKE}/wine.cmake")
endif()
if(DREAMGPU_BUILD_TOOLS)
  include("${DREAMGPU_CMAKE}/tools.cmake")
endif()
add_custom_target(dreamgpu-guest ALL DEPENDS ${DREAMGPU_RUNTIME_TARGETS} ${DREAMGPU_TOOL_TARGETS})
install(FILES "${DREAMGPU_ROOT}/LICENSE" DESTINATION licenses RENAME DREAMGPU.txt COMPONENT ${DREAMGPU_GUEST_OS})
install(DIRECTORY "${DREAMGPU_ROOT}/LICENSES/" DESTINATION licenses COMPONENT ${DREAMGPU_GUEST_OS})
install(FILES "${DREAMGPU_ROOT}/ATTRIBUTION.md" DESTINATION licenses COMPONENT ${DREAMGPU_GUEST_OS})
install(DIRECTORY "${DREAMGPU_ROOT}/support/attribution/" DESTINATION licenses/attribution COMPONENT ${DREAMGPU_GUEST_OS})
